CMAKE_MINIMUM_REQUIRED(VERSION 3.10)

FILE(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.c")

IF(RATELPROF_GPU_VENDOR STREQUAL "NVIDIA")
    # HSA based GPU profiling is replaced by the CUPTI backend (src/nvidia).
    LIST(FILTER SOURCES EXCLUDE REGEX "/src/ratelprof_ext\.c$")
    LIST(FILTER SOURCES EXCLUDE REGEX "/ratelprof_ext/ratelprof_completion_signal_pool\.c$")
    LIST(FILTER SOURCES EXCLUDE REGEX "/domains/(gpu_hsa|prof_hsa|roctx)_traced_functions\.c$")
ELSE()
    LIST(FILTER SOURCES EXCLUDE REGEX "/src/nvidia/")
ENDIF()

ADD_LIBRARY(${EXT_PROJECT_NAME}-lib-dyn     SHARED ${SOURCES})
ADD_LIBRARY(${EXT_PROJECT_NAME}-lib-static  STATIC ${SOURCES})

TARGET_LINK_LIBRARIES(${EXT_PROJECT_NAME}-lib-dyn    PUBLIC ${CORE_PROJECT_NAME}-lib-dyn 
                                                            Threads::Threads ${CMAKE_DL_LIBS})
TARGET_LINK_LIBRARIES(${EXT_PROJECT_NAME}-lib-static PUBLIC Threads::Threads ${CMAKE_DL_LIBS})

TARGET_INCLUDE_DIRECTORIES(${EXT_PROJECT_NAME}-lib-dyn      PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
TARGET_INCLUDE_DIRECTORIES(${EXT_PROJECT_NAME}-lib-static   PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
TARGET_INCLUDE_DIRECTORIES(${EXT_PROJECT_NAME}-lib-static   PUBLIC
    $<TARGET_PROPERTY:${CORE_PROJECT_NAME}-lib-static,INTERFACE_INCLUDE_DIRECTORIES>
)

IF(RATELPROF_GPU_VENDOR STREQUAL "NVIDIA")
    TARGET_LINK_LIBRARIES(${EXT_PROJECT_NAME}-lib-dyn    PUBLIC ${RATELPROF_CUPTI_LIBRARY})
    TARGET_LINK_LIBRARIES(${EXT_PROJECT_NAME}-lib-static PUBLIC ${RATELPROF_CUPTI_LIBRARY})
ENDIF()

SET_TARGET_PROPERTIES(${EXT_PROJECT_NAME}-lib-dyn       PROPERTIES OUTPUT_NAME ${EXT_PROJECT_NAME})
SET_TARGET_PROPERTIES(${EXT_PROJECT_NAME}-lib-static    PROPERTIES OUTPUT_NAME ${EXT_PROJECT_NAME})

CONFIGURE_FILE(
    ${CMAKE_CURRENT_SOURCE_DIR}/include/ratelprof_version.h.in
    ${CMAKE_CURRENT_SOURCE_DIR}/include/ratelprof_version.h
    @ONLY
)
