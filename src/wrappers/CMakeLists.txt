CMAKE_MINIMUM_REQUIRED(VERSION 3.10)

FILE(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.c")

IF(RATELPROF_GPU_VENDOR STREQUAL "NVIDIA")
    LIST(APPEND SOURCES "${RATELPROF_NV_GEN_DIR}/omp_tgt_rtl_methods.c")
    # CUDA Runtime/Driver are traced with CUPTI callbacks (no interposition) and
    # NVTX through the CUPTI NVTX injection.
    LIST(FILTER SOURCES EXCLUDE REGEX "/(hip|hsa|rocblas|roctx)_wrapper\.c$")
    IF(RATELPROF_HAVE_CUBLAS)
        LIST(APPEND SOURCES "${RATELPROF_NV_GEN_DIR}/cublas_wrapper.c")
    ENDIF()
ENDIF()

ADD_LIBRARY(${WRAPPER_PROJECT_NAME}-lib-dyn    SHARED ${SOURCES})
ADD_LIBRARY(${WRAPPER_PROJECT_NAME}-lib-static STATIC ${SOURCES})

TARGET_LINK_LIBRARIES(${WRAPPER_PROJECT_NAME}-lib-dyn       ${CORE_PROJECT_NAME}-lib-dyn    ${EXT_PROJECT_NAME}-lib-dyn)

SET_TARGET_PROPERTIES(${WRAPPER_PROJECT_NAME}-lib-dyn       PROPERTIES OUTPUT_NAME ${WRAPPER_PROJECT_NAME})
SET_TARGET_PROPERTIES(${WRAPPER_PROJECT_NAME}-lib-static    PROPERTIES OUTPUT_NAME ${WRAPPER_PROJECT_NAME})

    $<TARGET_PROPERTY:${CORE_PROJECT_NAME}-lib-static,INTERFACE_INCLUDE_DIRECTORIES>
    $<TARGET_PROPERTY:${EXT_PROJECT_NAME}-lib-static,INTERFACE_INCLUDE_DIRECTORIES>
)
